# Known interactive prompts, as data.
#
# Format: tab-separated, one prompt per line.
#   tool<TAB>kind<TAB>regex
#
# kind is one of: shell | repl | menu | password | fullscreen
#
# Ordering matters: the first match wins, so specific prompts must come before
# the generic "ends in $/#/%/>" catches near the bottom. Edit this file to add
# a prompt; no code change is needed.
#
# The regex is matched against the last non-blank line of a captured pane
# (re.search, so anchors are up to you).
#
# NOTE: prompts nest. `msf6 >` can become `msf6 exploit(..) >`, then a session
# gives `meterpreter >`, then `meterpreter > shell` gives `C:\>`. The current
# prompt therefore belongs to the *deepest* process in the pane, and must be
# rescanned when the process chain changes.

# ------------------------------------------------------------------ REPLs ---
python-repl	repl	^In \[\d+\]:\s*
python-repl	repl	^\s*>>>\s*
python-repl	repl	^\s*\.\.\.\s*
ipython-cont	repl	^\s{3,}\.\.\.:\s*
ruby-irb	repl	^irb\(main\):\d+:\d+[*>]\s*
ruby-irb	repl	^\d+:\d+[*>]\s*
ruby-irb	repl	irb\(main\):\d+:\d+[*>]\s*$
perl-debug	repl	^\s*DB<\d+>\s*
php-a	repl	^php >\s*
node-repl	repl	^>\s*$

# -------------------------------------------------------------- hypervisors ---
# QEMU's monitor, reached with Ctrl-a c on a nixos-shell/build-vm console. It is
# a real interactive prompt: commands like `quit`, `info status`, `sendkey` go
# here, and sending a shell command would be an error. Distinct from the
# guest's own shell on the same pane.
qemu-monitor	repl	^\(qemu\)\s*
qemu-monitor	repl	^QEMU [0-9.]+ monitor - type 'help'
# GRUB's menu/editor, which a VM can drop to before any guest shell exists.
grub-shell	shell	^grub>\s*$
grub-edit	menu	^grub edit>\s*$

# ----------------------------------------------------------------- shells ---
root-shell	shell	^[^\s]*#\s*$
root-shell	shell	^[^\s]*➜\s*\S*\s*$
user-shell	shell	^[^\s]*\$\s*$
user-shell	shell	^\[[^]]+\][#$]\s*
user-shell	shell	^[^\s]*%\s*$
# `user@host <cwd> >` -- the shape stock bash/zsh produce with a bare `\u@\h:\w\$`
# PS1, i.e. what a NixOS VM or a nixos-shell guest actually shows:
#
#   root@isp-cobalt:~/ >
#
# The generic catches above and below use `^[^\s]*` as the whole prefix, so a
# prompt with a SPACE in it before the final sigil matches none of them. That is
# not cosmetic: `pane_state` then reports "hosts a terminal; no prompt on
# screen" for a guest sitting at its prompt, `is_at_guest_shell` answers False,
# and every HAT verb refuses to type -- measured 2026-11-11 on the cobalt
# fixture, where the prompt was on screen and `_stable_prompt` agreed it was
# stable while the table matched nothing.
user-at-host-shell	shell	^[^\s@]+@[^\s]+\s+\S*\s*[>#$%]\s*$
fish-shell	shell	^[^\s]*>\s*$
shell	shell	^[^\s]*[>#$%]\s*$
